\documentclass[tikz,border=3pt]{standalone}
\usepackage{amsmath,amssymb,bm}
\usetikzlibrary{arrows.meta,positioning,calc,fit}

\colorlet{agentA}{red!80!black}     % agent 1: policy, prompts, responses, loss
\colorlet{agentB}{blue!80!black}    % agent 2
\colorlet{learned}{orange!80!black} % learned evaluators: critic, reward model
\colorlet{update}{green!50!black}   % gradients and parameter updates

\tikzset{
  >={Stealth[length=3.2pt,width=3pt]},
  box/.style={draw, line width=1pt, inner sep=2.5pt, minimum height=13pt},
  policy/.style={box, rounded corners=3pt, minimum width=30pt},
  flow/.style={->, line width=.8pt},
  note/.style={font=\scriptsize, align=left},
  label/.style={anchor=west, font=\footnotesize},
}

% The MAREINFORCE family at one turn, left to right: the agents' sampled responses
% share one return per joint sample, and the same advantage weights each agent's
% own policy gradient. Columns are animation stages.
\begin{document}
\begin{tikzpicture}[x=1pt, y=1pt, font=\footnotesize]
  % Stage 0: each agent's policy acts on its own history.
  \node[agentA] (h1) at (0,18) {$h_{1,t}$};
  \node[agentB] (h2) at (0,-18) {$h_{2,t}$};
  \node[policy, agentA] (p1) at (36,18) {$\pi_{\theta_1}$};
  \node[policy, agentB] (p2) at (36,-18) {$\pi_{\theta_2}$};
  \draw[flow, agentA] (h1) -- (p1);
  \draw[flow, agentB] (h2) -- (p2);

  % Stage 1: the k-th responses form a joint sample with one shared return.
  \node[box] (R) at (100,0) {$R^k_t$};
  \node[note] at (100,31) {shared return};
  \node[note] at (100,-13) {$k=1,\ldots,K$};
  \draw[flow, agentA] (p1.east) -- node[above] {$a^k_{1,t}$} (72,18) |- ([yshift=3pt]R.west);
  \draw[flow, agentB] (p2.east) -- node[below] {$a^k_{2,t}$} (72,-18) |- ([yshift=-3pt]R.west);

  % Stage 2: a baseline turns the returns into advantages.
  \node[box] (A) at (171,0) {$\hat A^k_t=R^k_t-b(\mathbf{h}_t)$};
  \node[note] at (171,31) {advantage};
  \draw[flow] (R) -- (A);

  % Stage 3: the same advantage weights each agent's own policy gradient.
  \node[box, agentA] (g1) at (286,18)
    {$\textcolor{black}{\hat A^k_t}\,\nabla_{\theta_1}\log\pi_{\theta_1}(a^k_{1,t}\mid h_{1,t})$};
  \node[box, agentB] (g2) at (286,-18)
    {$\textcolor{black}{\hat A^k_t}\,\nabla_{\theta_2}\log\pi_{\theta_2}(a^k_{2,t}\mid h_{2,t})$};
  \node[note] at (286,31) {policy gradient};
  \draw[flow] ([yshift=3pt]A.east) -- (222,3) |- (g1.west);
  \draw[flow] ([yshift=-3pt]A.east) -- (222,-3) |- (g2.west);
\end{tikzpicture}
\end{document}
